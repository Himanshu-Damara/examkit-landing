import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";

// Number of salt rounds used for hashing
const SALT_ROUNDS = 10;

// Secret key for signing and verifying tokens
const AUTH_SECRET_KEY = "SUPER_SECRET_SECUREAUTH_KEY";

// Hash a plain-text password
function hashPassword(password) {
    const salt = bcrypt.genSaltSync(SALT_ROUNDS);
    const hash = bcrypt.hashSync(password, salt);
    return hash;
}

// Verify a plain-text password against a hash
function verifyPassword(password, hash) {
    return bcrypt.compareSync(password, hash);
}

// Generate a session token (JWT)
function generateToken(userId, email) {
    const payload = {
        userId,
        email
    };

    const token = jwt.sign(payload, AUTH_SECRET_KEY, {
        expiresIn: "1h"
    });

    return token;
}

// Decode a session token (JWT)
function decodeToken(token) {
    try {
        const decoded = jwt.verify(token, AUTH_SECRET_KEY);
        return decoded;
    } catch (error) {
        return null;
    }
}

// Do not modify this
// --- Example Usage ---
if (process.env.NODE_ENV !== "test") {
   const hashed = hashPassword("mySecurePass123");
   console.log("Hashed Password:", hashed);

   console.log("Correct Password Check:", verifyPassword("mySecurePass123",
hashed));
   console.log("Wrong Password Check:", verifyPassword("wrongPass", hashed));

   const myToken = generateToken("U100", "user@secureauth.com");
   console.log("Generated Token:", myToken);

   console.log("Decoded Token Payload:", decodeToken(myToken));
   console.log("Invalid Token:", decodeToken("fake.token.value"));
}

// Export for testing
export { hashPassword, verifyPassword, generateToken, decodeToken,
AUTH_SECRET_KEY };

/**
* --------------------------------------------------
* TASK 5: verifyToken Middleware
* --------------------------------------------------
*/
function verifyToken(req, res, next) {
 const authHeader = req.headers.authorization;

 if (!authHeader || !authHeader.startsWith("Bearer ")) {
   return res.status(401).json({
     message: "No token provided"
   });
 }

 const token = authHeader.split(" ")[1];

 try {
   const decoded = jwt.verify(token, JWT_SECRET);
   req.user = decoded;
   next();
 } catch (error) {
   if (error.name === "TokenExpiredError") {
     return res.status(401).json({
       message: "Token expired"
     });
   }

   return res.status(403).json({
     message: "Invalid token"
   });
 }
}

/**
* --------------------------------------------------
* TASK 3: Register a User
* --------------------------------------------------
* POST /api/register
*/
app.post("/api/register", (req, res) => {
 try {
   const { username, email, password } = req.body;

   // Check required fields
   if (!username || !email || !password) {
     return res.status(400).json({
       message: "Username, email and password are required"
     });

   // Check duplicate username
   const existingUser = users.find(
     user => user.username === username
   );

   if (existingUser) {
     return res.status(409).json({
       message: "User already exists"

   const passwordHash = hashPassword(password);

   // Create user
   const newUser = {
     id: nextId,
     username,
     email,
     passwordHash
   };

   // Increment ID
   nextId++;

   // Store user
   users.push(newUser);

   // Return user without password hash
   return res.status(201).json({
     message: "User registered successfully",
     user: {
       id: newUser.id,
       username: newUser.username,
       email: newUser.email

// Secret key for signing and verifying tickets
const TICKET_SECRET_KEY = "SUPER_SECRET_EVENTMASTER_KEY";

// Generate a digital ticket (JWT)
function generateTicket(attendeeId, eventName) {
   const payload = {
       attendeeId,
       eventName
   };

   const token = jwt.sign(
       payload,
       TICKET_SECRET_KEY,
       { expiresIn: "24h" }
   );

   return token;
}

// Verify a digital ticket (JWT)
function verifyTicket(ticket) {
   try {
       const decoded = jwt.verify(
           ticket,
           TICKET_SECRET_KEY
       );

       return decoded;
   } catch (error) {
       return null;
   }
}

// Do not modify this

   // example usage
   const myTicket = generateTicket("12345", "Rock Concert 2025");
   console.log("Generated Ticket:", myTicket);

   const verified = verifyTicket(myTicket);
   console.log("Verified Ticket Payload:", verified);

   const invalid = verifyTicket("fake.token.value");
   console.log("Invalid Ticket:", invalid);
}

function parseStudent(body) {
 return {
   name: String(body.name ?? "").trim(),
   roll_number: String(body.roll_number ?? "").trim(),
   department: String(body.department ?? "").trim(),
   cgpa: Number(body.cgpa),
   year: Number(body.year)
 };
}

function isValidStudent(student) {
 return (
   student.name.length > 0 &&
   student.roll_number.length > 0 &&
   student.department.length > 0 &&
   Number.isFinite(student.cgpa) &&
   student.cgpa >= 0 &&
   student.cgpa <= 10 &&
   Number.isInteger(student.year) &&
   student.year >= 1
 );
}

     .collection("students")

     .sort({ name: 1 })

 } catch (err) {
   res.status(500).json({ error: err.message });
 }
});

router.get("/search", async (req, res) => {
 try {
   const q = String(req.query.q || "").trim();

   const safe = q.replace(/[.*+?^${}()|[]\]/g, "\$&");

   const filter = q
     ? {
         $or: [
           { name: { $regex: safe, $options: "i" } },
           { roll_number: { $regex: safe, $options: "i" } },
           { department: { $regex: safe, $options: "i" } }
         ]
       }
     : {};

   const students = await getDB()
     .collection("students")
     .find(filter)
     .sort({ name: 1 })
     .toArray();

   res.json(students);
 } catch (err) {
   res.status(500).json({ error: err.message });
 }
});

    // Handle non-200 responses

    // Load HTML into Cheerio
    const data = await response.text();
    const $ = cheerio.load(data);

    // Extract BASIC INFO
    const name = $(".user-details-container h1").first().text().trim() ||
"N/A";

    // Extract Affiliation (Institution / Organisation)
    let institution = "";
    let organisation = "";

    institution =
      $(".user-details-container")
        .find("li:contains('Institution')")
   .text()
   .replace(/Institution[:s]*/i, "")
   .trim() || "";

organisation =
 $(".user-details-container")
   .find("li:contains('Organisation')")
   .text()
   .replace(/Organisation[:s]*/i, "")
   .trim() || "";

const affiliation = institution || organisation || "N/A";

// Extract COUNTRY
const country =
 $(".user-details-container span[class*='country']").text().trim() ||
 $("li:contains('Country')")
   .text()
   .replace(/Country[:s]*/i, "")
   .trim() ||
 "N/A";

// Extract RATINGS & RANKS
const ratingText = $(".rating-number").text().trim();
const currentRating = ratingText ? parseInt(ratingText, 10) || 0 : 0;

const stars = $(".rating-star").text().trim() || "Unrated";

const globalRankText = $(".rating-ranks a")
 .first()
 .text()
 .replace(/D/g, "");

const globalRank = globalRankText
 ? parseInt(globalRankText, 10) || 0
 : 0;

const highestRating =
 highestRatingText && highestRatingText.match(/d+/)
   ? parseInt(highestRatingText.match(/d+/)[0], 10)
   : currentRating;

let totalSolved = 0;

$(".problems-solved .content h3").each((_, el) => {
 const match = $(el).text().match(/d+/);
 if (match) {
   totalSolved += parseInt(match[0], 10);
 }

let totalContests = 0;
const contestDiv = $("div.contest-participated-count");

if (contestDiv.length) {
 const bTag = contestDiv.find("b").first();

 if (bTag.length) {
   totalContests = parseInt(bTag.text().trim(), 10) || 0;

if (lpSection.length) {

 if (h3Tags.length) {

   if (learningMatch) {

   };

// TASK 4: Get active loans
router.get("/loans/active", async (req, res) => {
 try {
   const db = getDB();

   const loans = await db.collection("loans")
     .aggregate([
       {
         $match: {
           returned_on: null
         }
       },
       {
         $lookup: {
           from: "books",
           localField: "book_id",
           foreignField: "_id",
           as: "book"
         }
       },
       {
         $unwind: {
           path: "$book",
           preserveNullAndEmptyArrays: true
         }
       },
       {
         $addFields: {
           book_title: "$book.title",
           overdue: {
             $lt: ["$due_on", "$$NOW"]
           }
         }
       },
       {
         $project: {
           book: 0
         }
       },
       {
         $sort: {
           due_on: 1
         }
       }
     ])
     .toArray();

   res.status(200).json(loans);
 } catch (err) {
   res.status(500).json({ error: err.message });
 }
});

// TASK 2: Issue a book
router.post("/loans", async (req, res) => {
 try {
   const db = getDB();

   const { book_id, student_name, roll_number } = req.body;

   const books = db.collection("books");
   const loans = db.collection("loans");

   const bookObjectId = new ObjectId(book_id);

   // Claim one available copy atomically.
   const update = await books.updateOne(
     {
       _id: bookObjectId,
       available_copies: { $gt: 0 }
     },
     {
       $inc: { available_copies: -1 }
     }
   );
   if (update.modifiedCount === 0) {
     return res.status(409).json({
       error: "No copies available"
     });
   }

   const issued_on = new Date();
   const due_on = new Date(issued_on);
   due_on.setDate(due_on.getDate() + 14);

   const loan = {
     book_id: bookObjectId,
     student_name,
     roll_number,
     issued_on,
     due_on,
     returned_on: null
   };

   try {
     const result = await loans.insertOne(loan);

     return res.status(201).json({
       ...loan,
       _id: result.insertedId
     });
   } catch (error) {
     // Restore the copy if inserting the loan fails.
     await books.updateOne(
       { _id: bookObjectId },
       { $inc: { available_copies: 1 } }
     );
     throw error;
   }
 } catch (err) {
   res.status(500).json({ error: err.message });
 }
});

// TASK 3: Return a book
router.put("/loans/:id/return", async (req, res) => {
 try {
   const db = getDB();

   const loans = db.collection("loans");
   const books = db.collection("books");

   const loanId = new ObjectId(req.params.id);

   const loan = await loans.findOne({
     _id: loanId
   });

   if (!loan) {
      return res.status(404).json({
        error: "Loan not found"
      });
    }

    if (loan.returned_on !== null) {
      return res.status(409).json({
        error: "Already returned"
      });
    }

    const returned_on = new Date();

    const update = await loans.updateOne(
      {
        _id: loanId,
        returned_on: null
      },
      {
        $set: { returned_on }
      }
    );

    if (update.modifiedCount === 0) {
      return res.status(409).json({
        error: "Already returned"
      });
    }

    await books.updateOne(
      { _id: loan.book_id },
      { $inc: { available_copies: 1 } }
    );

    const updatedLoan = await loans.findOne({
      _id: loanId
    });

    return res.status(200).json(updatedLoan);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});
